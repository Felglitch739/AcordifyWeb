interface LyricsParams {
  // Musical context (injected from app state, not user-controlled)
  activeChords: string[]       // e.g. ["Cmaj7","Am9","Dm7","G13"]
  keyRoot: string              // e.g. "A"
  mode: 'major' | 'minor'
  bpm: number                  // e.g. 90
  mood: string                 // e.g. "Jazz Melancólico"

  // User-controlled parameters
  rhymeScheme: 'ABAB' | 'AABB' | 'ABBA' | 'free'
  emotionalMood: number        // 0-100, 0=melancólico, 100=esperanzador
  narrativePerson: '1ra' | '2da' | '3ra'
  metaphorDensity: 'literal' | 'balanced' | 'poetic'
  thematicConcept: string      // free text, optional, max 60 chars
  language: 'es' | 'en'
  linesToGenerate: 4 | 8
}

interface LyricsResult {
  lines: Array<{
    text: string
    chord: string              // chord active on this line
    syllableCount: number
    rhymeLabel: string         // "A" | "B" | "C"
  }>
  detectedScheme: string
  emotionalTag: string         // e.g. "resignación melancólica"
  chordProOutput: string       // ready to inject into ChordPro Notebook
}

SYSTEM PROMPT TO BUILD (inside lyricsService.ts):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

The buildLyricsPrompt(params: LyricsParams) function must
construct a system + user prompt that:

- Maps emotionalMood (0-100) to descriptive language:
  0-20   → "profundamente melancólico, casi nihilista"
  21-40  → "melancólico con destellos de aceptación"
  41-60  → "ambivalente, entre la sombra y la luz"
  61-80  → "esperanzador pero realista"
  81-100 → "luminoso, resolutivo, esperanza plena"

- Maps metaphorDensity:
  literal   → "lenguaje directo, imágenes concretas, sin metáforas"
  balanced  → "metáforas moderadas, imágenes vívidas pero accesibles"
  poetic    → "lenguaje altamente metafórico, simbolismo denso"

- Maps BPM to rhythmic feel hint:
  < 70 BPM  → "tempo lento, sílabas largas, pocas palabras por línea"
  70-100    → "tempo moderado, balance entre fluidez y peso"
  > 100 BPM → "tempo rápido, líneas densas, ritmo interno marcado"

- Enforces syllabic consistency based on BPM mapping above
- Maps each line to a chord from activeChords cyclically
- Output: JSON only, matching LyricsResult schema exactly
- Language of lyrics: {{LANGUAGE}}
- Prohibits: clichés de amor romántico genérico, relleno silábico,
  cambio de persona narrativa mid-stanza
